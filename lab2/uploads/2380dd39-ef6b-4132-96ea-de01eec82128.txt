#include <iostream>
#include <iomanip>

using namespace std;

int main() {
    int rows = 0, cols = 0;
    int **matrix;

    cout << "Enter the rows number: ";
    cin >> rows;

    cout << "Enter the columns number: ";
    cin >> cols;

    matrix = new int *[rows];

    for (int i = 0; i < rows; i++) {
        matrix[i] = new int[cols];
        for (int j = 0; j < cols; j++) {
            cout << "Enter the matrix element [" << i << "][" << j << "] = ";
            cin >> matrix[i][j];
        }
    }

    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            cout << setw(3) << matrix[i][j];
        }
        cout << endl;
    }

    for (int i = 0; i < rows; i++) {
        delete[] matrix[i];
    }

    delete[] matrix;

    return 0;
}
